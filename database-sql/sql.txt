create table dish
(
    id          bigint auto_increment comment '主键'
        primary key,
    name        varchar(32)      not null comment '菜品名称',
    category_id bigint           not null comment '菜品分类id',
    price       decimal(10, 2)   null comment '菜品价格',
    image       varchar(255)     null comment '图片',
    description varchar(255)     null comment '描述信息',
    status      bigint default 1 null comment '启用状态:0 停售 1 起售',
    create_time datetime         null comment '创建时间',
    update_time datetime         null comment '更新时间',
    create_user bigint           null comment '创建人',
    update_user bigint           null comment '修改人',
    constraint idx_dish_name
        unique (name)
)
    comment '菜品' collate = utf8mb3_bin;

create table dish_detail
(
    id         bigint auto_increment comment '主键'
        primary key,
    dish_id    bigint       not null comment '菜品',
    key_type   varchar(32)  null comment '标签',
    value_list varchar(255) null comment '标签详细，用,隔开'
)
    comment '菜品口味关系表' collate = utf8mb3_bin;

create table `order`
(
    id                      bigint auto_increment comment '主键'
        primary key,
    status                  bigint default 1 not null comment '订单状态
1待支付：下单未付款
2待商家接单：已付款，商家还没接单
3商家接单,制作中：商家确认接单，正在做菜
4待骑手取餐：商家出餐完成，骑手还没到店
5骑手已取餐，配送中：骑手拿到餐，在路上，实时看定位
6骑手已送达：骑手点送达，等待用户确认
7订单已完成：系统自动确认收货
8订单已取消：未接单退款、商家拒单、超时取消、售后全额退款',
    user_id                 bigint           not null comment '下单用户',
    username                varchar(32)      null comment '用户名称',
    consignee               varchar(32)      null comment '收货人',
    phone                   varchar(11)      null comment '手机号',
    address_id              bigint           not null comment '地址id',
    address                 varchar(255)     null comment '地址',
    pay_status              bigint default 0 not null comment '关联orderPay',
    remark                  varchar(100)     null comment '备注',
    delivery_status         mediumtext       null comment '配送方式：1立即送出  0选择具体时间',
    start_delivery_time     datetime         null comment '配送时间',
    estimated_delivery_time datetime         null comment '预计送达时间',
    delivery_time           datetime         null comment '送达时间'
)
    comment '订单表' collate = utf8mb3_bin;

(

    pay_status       bigint default 0 not null comment '支付状态 0未支付 1已支付 2退款',
    checkout_time    datetime         null comment '结账时间',
    amount           decimal(10, 2)   not null comment '实收金额',
    remark           varchar(100)     null comment '备注',
    cancel_reason    varchar(255)     null comment '订单取消原因',
    rejection_reason varchar(255)     null comment '订单拒绝原因',
    cancel_time      datetime         null comment '取消时间',
    create_time      datetime         null comment '创建时间',
    update_time      datetime         null comment '更新时间',
    create_user      bigint           null comment '创建人',
    update_user      bigint           null comment '修改人'
)
    comment '订单状态';
